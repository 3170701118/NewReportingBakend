## CHAMBER 式数据
```sql
SELECT a.LINE, SUBSTR(a.EQ_ID, 1, 5) AS EQP_ID,
	  MAX(DECODE(SIGN(INSTR(a.EQ_ID, '.')), 1, 1, 0)) AS CHAMBER
 FROM EQDM.EQ_UDATA a, EQDM.EQUIPMENT_ATTRIBUTES b
WHERE a.LINE = b.LINE(+) AND SUBSTR(a.EQ_ID, 1, 5) = b.EQP_ID(+) AND a.CHAMBER_FLAG = '1'
  AND b.OWNER IS NOT NULL AND b.OWNER <> 'ADMIN'
  AND b.ISACTIVE = '1' AND b.RECORD_STATUS = '1'
--  AND AA.EQP_ID = SUBSTR(a.EQ_ID, 1, 5) // AA修改为对应表
GROUP BY a.LINE, SUBSTR(a.EQ_ID, 1, 5)
HAVING MAX(DECODE(SIGN(INSTR(a.EQ_ID, '.')), 1, 1, 0)) = 1
```

### 机台个状态的时间 SBY PRD SDT UDT + 状态code 1000 1200 等
```sql
SELECT X.OWNER,X.EQP_ID,X.CHAMBER_ID,X.EVENTSTARTTMST,X.EVENTENDTMST,X.PREV_E10_STATE_CODE,X.PREV_STATE,X.SHIFT_START_DATE,X.SHIFT_END_DATE,X.STATUS_CLAIM_USER,Y.SHIFT_DATE,
CASE WHEN X.SHIFT_START_DATE=X.SHIFT_END_DATE THEN ROUND(X.KEEP_TIME*24,2)
WHEN Y.SHIFT_DATE=X.SHIFT_START_DATE THEN ROUND((Y.SHIFT_TIME+1-X.EVENTSTARTTMST)*24,2)
WHEN Y.SHIFT_DATE=X.SHIFT_END_DATE THEN ROUND((X.EVENTENDTMST-Y.SHIFT_TIME)*24,2) ELSE 24 END KEEP_TIME
FROM(
    SELECT B.OWNER,B.EQP_ID,A.CHAMBER_ID,A.EVENTSTARTTMST,NVL(A.EVENTENDTMST,SYSDATE)EVENTENDTMST,A.PREV_E10_STATE_CODE,A.PREV_STATE,A.SHIFT_START_DATE,
    NVL(A.SHIFT_END_DATE,TO_CHAR(SYSDATE-7.5/24,'YYYYMMDD')) SHIFT_END_DATE,A.STATUS_CLAIM_USER,
    NVL(A.EVENTENDTMST,SYSDATE)-A.EVENTSTARTTMST KEEP_TIME
    FROM EQDM.CHAMBER_STATUS_TRACKING A, 
    EQDM.EQUIPMENT_ATTRIBUTES B
    WHERE A.EQUIPMENT_KEY = B.EQUIPMENT_KEY 
    AND B.AREA_ID IS NOT NULL AND OWNER='ETCH' 
    AND B.EQP_GROUP IS NOT NULL
    AND (A.CHAMBER_ID LIKE 'EM%'OR A.CHAMBER_ID LIKE 'EU%'OR A.CHAMBER_ID LIKE 'EG%'OR A.CHAMBER_ID LIKE 'EC%'OR A.CHAMBER_ID LIKE 'EK%')
    AND A.EVENTSTARTTMST>TO_DATE(TO_CHAR(SYSDATE-7,'yyyy/mm/dd')||'07:30','yyyy/mm/dd hh24:mi') 
    --AND A.EVENTENDTMST<TO_DATE(TO_CHAR(SYSDATE,'yyyy/mm/dd')||'07:30','yyyy/mm/dd hh24:mi')
    --AND A.CHAMBER_ID='EMA19.B' 
)X,
(
    SELECT TO_CHAR(TO_DATE('20220817','YYYYMMDD')+ROWNUM-1,'YYYYMMDD') SHIFT_DATE,TO_DATE('20220817','YYYYMMDD')+ROWNUM-1+7.5/24 SHIFT_TIME
    FROM DUAL
    CONNECT BY ROWNUM<=TRUNC(SYSDATE)-TO_DATE('20220817','YYYYMMDD')+1
)Y
WHERE 1=1 AND Y.SHIFT_DATE BETWEEN X.SHIFT_START_DATE AND X.SHIFT_END_DATE
ORDER BY CHAMBER_ID,EVENTENDTMST DESC

WITH RecursiveLot (parent_lot_id, lot_id, original_lot_id, level_) AS (
  SELECT parent_lot_id, lot_id, lot_id original_lot_id, 1 level_
  FROM ods.lot 
  WHERE parent_lot_id IS NOT NULL AND PW_FLAG =1
  UNION ALL
  SELECT t.parent_lot_id, t.lot_id, r.original_lot_id, r.level_ + 1
  FROM ods.lot  t
  JOIN RecursiveLot r ON t.lot_id = r.parent_lot_id
)
SELECT original_lot_id, lot_id, parent_lot_id, level_
FROM RecursiveLot
WHERE lot_id IS NOT NULL
```
